# 07 – RealLeafletMapView.kt  +  assets/map.html
**المسارات:** `ui/screens/map/RealLeafletMapView.kt` و `app/src/main/assets/map.html`  |  **الإجراء:** تعديلات موضعية.

## المشاكل المكتشفة (حقيقية في الكود المرفوع)
1. **عقد الربط JS ↔ Kotlin مكسور:**
   - Kotlin يرسل المفاتيح `type`/`name`/`emoji`/`badgeColor`، وJS يقرأ `category`/`title` → كل الدبابيس بنفس الرمز 📍 وعنوان "undefined".
   - JS ينادي `AndroidBridge.onMarkerSelected` بينما Kotlin يعرّف `onMarkerClicked` → **الضغط على أي دبوس في الخريطة الحقيقية لا يصل للتطبيق** (لا نافذة سفلية ولا ملاحة).
2. **حقن HTML**: `m.title` يُدرج في HTML بلا تهريب، وأسماء المزودين من المستخدمين، والـ WebView فيه `AndroidBridge` → ثغرة أمنية.
3. دوال الجسر (`@JavascriptInterface`) تعمل على خيط خلفي وتغيّر حالة Compose وتنفّذ Toast/UI → غير آمن.
4. الجسر يلتقط `nearbyProviders` والـ callbacks لحظة الإنشاء فقط (قوائم قديمة بعد إعادة التركيب).
5. الرجوع الاحتياطي `window.setMarkers('$markersJson')` بلا تهريب يكسر JS عند وجود `'` في الأسماء.
6. لا يوجد رسم للمسار ولا محطات مرقمة ولا تتبع للمستخدم.
7. `WebView` لا يُدمَّر عند الخروج (تسريب ذاكرة)، و`allowFileAccess=true` غير لازم (الأصول `file:///android_asset` لا تتأثر).
8. لا اعتراض للبلاطات → لا أوفلاين (المستند 08).

```javascript
function esc(s) {
    return String(s == null ? '' : s).replace(/[&<>"']/g, function (c) {
        return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
    });
}
function setMarkersList(markers) {
    if (!markerClusterGroup) return;
    markerClusterGroup.clearLayers();
    markers.forEach(function (m) {
        if (typeof m.lat !== 'number' || typeof m.lng !== 'number' || (m.lat === 0 && m.lng === 0)) return;
        var type = String(m.type || '').toUpperCase();
        var color = /^#[0-9A-Fa-f]{3,8}$/.test(m.badgeColor || '') ? m.badgeColor : '#2563EB';
        var emoji = m.emoji || '📍';
        var approx = m.isApprox === true;
        var html = '<div class="custom-pin-container">' +
            '<div class="custom-pin-head" style="background-color:' + color + ';' +
            (approx ? 'opacity:0.65;border:2px dashed #fff;' : '') + '">' + esc(emoji) + '</div>' +
            '<div class="custom-pin-label">' + esc(m.name) + (approx ? ' (موقع تقريبي)' : '') + '</div></div>';
        var icon = L.divIcon({ className: 'custom-leaflet-marker', html: html, iconSize: [145, 65], iconAnchor: [72, 38] });
        var marker = L.marker([m.lat, m.lng], { icon: icon });
        marker.on('click', function () { onMarkerClick(String(m.id), type); });
        markerClusterGroup.addLayer(marker);
    });
}
function onMarkerClick(id, type) {
    if (window.AndroidBridge && window.AndroidBridge.onMarkerClicked) {
        window.AndroidBridge.onMarkerClicked(id, type);
    }
}
```

### (ج) أضف الدوال التالية قبل `window.onload` (المسار + المحطات + دبوس البحث):
```javascript
var routeLayer = null, routeCasing = null, waypointLayer = null, searchPin = null;

function clearRoute() {
    if (map && routeLayer) { map.removeLayer(routeLayer); routeLayer = null; }
    if (map && routeCasing) { map.removeLayer(routeCasing); routeCasing = null; }
}
function setRouteBase64(b64, fit) {
    try {
        var pts = JSON.parse(window.atob(b64));            // [[lat,lng],...]
        clearRoute();
        if (!map || !pts || pts.length < 2) return;
        routeCasing = L.polyline(pts, { color: '#0B1120', weight: 10, opacity: 0.9, lineCap: 'round', lineJoin: 'round' }).addTo(map);
        routeLayer = L.polyline(pts, { color: '#00E5FF', weight: 5, opacity: 1, lineCap: 'round', lineJoin: 'round' }).addTo(map);
        if (fit) map.fitBounds(routeLayer.getBounds(), { padding: [60, 60] });
    } catch (e) { console.error('setRoute error', e); }
}
function setWaypointMarkersBase64(b64) {
    try {
        var list = JSON.parse(decodeURIComponent(escape(window.atob(b64))));   // [{label,lat,lng}]
        if (!map) return;
        if (waypointLayer) { map.removeLayer(waypointLayer); }
        waypointLayer = L.layerGroup();
        list.forEach(function (w, i) {
            var html = '<div style="background:#10B981;color:#fff;border:2px solid #fff;border-radius:50%;width:26px;height:26px;' +
                'display:flex;align-items:center;justify-content:center;font-weight:bold;font-size:12px;">' + (i + 1) + '</div>';
            var icon = L.divIcon({ className: 'wp-marker', html: html, iconSize: [26, 26], iconAnchor: [13, 13] });
            L.marker([w.lat, w.lng], { icon: icon, interactive: false }).addTo(waypointLayer);
        });
        waypointLayer.addTo(map);
    } catch (e) { console.error('setWaypointMarkers error', e); }
}
function showSearchPinBase64(b64) {
    try {
        var p = JSON.parse(decodeURIComponent(escape(window.atob(b64))));      // {name,lat,lng}
        if (!map) return;
        if (searchPin) { map.removeLayer(searchPin); searchPin = null; }
        var html = '<div class="custom-pin-container"><div class="custom-pin-head" style="background-color:#F59E0B;">📍</div>' +
            '<div class="custom-pin-label">' + esc(p.name) + '</div></div>';
        var icon = L.divIcon({ className: 'custom-leaflet-marker', html: html, iconSize: [145, 65], iconAnchor: [72, 38] });
        searchPin = L.marker([p.lat, p.lng], { icon: icon }).addTo(map);
        map.setView([p.lat, p.lng], 17);
    } catch (e) { console.error('showSearchPin error', e); }
}
function clearSearchPin() { if (map && searchPin) { map.removeLayer(searchPin); searchPin = null; } }
```
> الدوال معرّفة كـ `function` عامة فهي متاحة كـ `window.xxx` (كما تفعل بقية الدوال الحالية). لا تُدخلها داخل IIFE.

```kotlin
addJavascriptInterface(object {
    @JavascriptInterface fun onMapReady() { mainHandler.post { isMapReady = true } }
    @JavascriptInterface fun onMapClicked() { mainHandler.post { onDeselectState() } }
    @JavascriptInterface fun onUserPanned() { mainHandler.post { onUserPannedState?.invoke() } }
    @JavascriptInterface fun onMarkerClicked(id: String, type: String) {
        mainHandler.post {
            when (type.uppercase()) {
                "PROVIDER" -> providersState.find { it.id == id }?.let { onProviderSelectedState(it) }
                "STORE", "RESTAURANT", "MEDICAL" -> storesState.find { it.id == id }?.let { onStoreSelectedState(it) }
                "PROPERTY" -> propertiesState.find { it.id == id }?.let { onPropertySelectedState(it) }
            }
        }
    }
    @JavascriptInterface fun onMapModeChanged(newMode: String) { android.util.Log.d("LeafletBridge", "Map mode: $newMode") }
    @JavascriptInterface fun getAvailableLayers(): String = "[\"OSM\",\"HOT\",\"TOPO\",\"SATELLITE\",\"VELA_3D\"]"
}, "AndroidBridge")
```

### (ط) أضف LaunchedEffects جديدة (بجانب الموجودة):
```kotlin
// رسم المسار (يُبنى الـ JSON خارج الخيط الرئيسي، مع تخفيف النقاط إلى ≈2000)
LaunchedEffect(isMapReady, routeCoordinates, routeKey) {
    if (!isMapReady || webViewRef == null) return@LaunchedEffect
    if (routeCoordinates.size < 2) {
        webViewRef?.evaluateJavascript("if (window.clearRoute) { window.clearRoute(); }", null)
        return@LaunchedEffect
    }
    val b64 = withContext(Dispatchers.Default) {
        val step = maxOf(1, routeCoordinates.size / 2000)
        val arr = JSONArray()
        var i = 0
        while (i < routeCoordinates.size) {
            arr.put(JSONArray().put(routeCoordinates[i].first).put(routeCoordinates[i].second)); i += step
        }
        val last = routeCoordinates.last()
        arr.put(JSONArray().put(last.first).put(last.second))
        Base64.encodeToString(arr.toString().toByteArray(Charsets.UTF_8), Base64.NO_WRAP)
    }
    webViewRef?.evaluateJavascript("if (window.setRouteBase64) { window.setRouteBase64('$b64', ${!followUser}); }", null)
}

// محطات مرقمة
LaunchedEffect(isMapReady, waypointMarkers) {
    if (!isMapReady || webViewRef == null) return@LaunchedEffect
    val arr = JSONArray()
    waypointMarkers.forEach { (label, lat, lng) -> arr.put(JSONObject().put("label", label).put("lat", lat).put("lng", lng)) }
    val b64 = Base64.encodeToString(arr.toString().toByteArray(Charsets.UTF_8), Base64.NO_WRAP)
    webViewRef?.evaluateJavascript("if (window.setWaypointMarkersBase64) { window.setWaypointMarkersBase64('$b64'); }", null)
}

// دبوس نتيجة البحث
LaunchedEffect(isMapReady, focusPoint) {
    if (!isMapReady || webViewRef == null) return@LaunchedEffect
    if (focusPoint == null) {
        webViewRef?.evaluateJavascript("if (window.clearSearchPin) { window.clearSearchPin(); }", null)
    } else {
        val o = JSONObject().put("name", focusPoint.first).put("lat", focusPoint.second).put("lng", focusPoint.third)
        val b64 = Base64.encodeToString(o.toString().toByteArray(Charsets.UTF_8), Base64.NO_WRAP)
        webViewRef?.evaluateJavascript("if (window.showSearchPinBase64) { window.showSearchPinBase64('$b64'); }", null)
    }
}
```
